# RUN: llvm-mc -triple tlcs900 -disassemble < %s | FileCheck %s
#
# Test disassembly of 8-bit direct addressing mode (D8 prefix).
# F0 = destination direct 8-bit.

# LDA XBC, 0x0a: [F0, 0x0A, 0x31]
# CHECK: lda_dd8l xbc, 10
0xf0 0x0a 0x31

# LDA XWA, 0x08: [F0, 0x08, 0x30]
# CHECK: lda_dd8l xwa, 8
0xf0 0x08 0x30

# LDA XDE, 0xFF: [F0, 0xFF, 0x32]
# CHECK: lda_dd8l xde, 255
0xf0 0xff 0x32

# ST_DD8B: LD (addr8), Rb: [F0, addr, 0x40+r]
# CHECK: st_dd8b a, 52
0xf0 0x34 0x41

# ST_DD8W: LD (addr8), Rw: [F0, addr, 0x50+r]
# CHECK: st_dd8w wa, 52
0xf0 0x34 0x50

# ST_DD8L: LD (addr8), Rl: [F0, addr, 0x60+r]
# CHECK: st_dd8l xbc, 10
0xf0 0x0a 0x61

# STIB_DD8: LD (addr8), #imm8: [F0, addr, 0x00, imm]
# CHECK: stib_d8 41, 64
0xf0 0x29 0x00 0x40

# STIW_DD8: LD (addr8), #imm16: [F0, addr, 0x02, lo, hi]
# CHECK: stiw_d8 41, 52, 18
0xf0 0x29 0x02 0x34 0x12

# RES bit, (addr8): [F0, addr, 0xB0+bit]
# CHECK: res_dd8 3, 52
0xf0 0x34 0xb3

# SET bit, (addr8): [F0, addr, 0xB8+bit]
# CHECK: set_dd8 0, 52
0xf0 0x34 0xb8

# CHECK: set_dd8 7, 255
0xf0 0xff 0xbf

# BIT bit, (addr8): [F0, addr, 0xC8+bit]
# CHECK: bit_dd8 5, 100
0xf0 0x64 0xcd

# CHG bit, (addr8): [F0, addr, 0xC0+bit]
# CHECK: chg_dd8 2, 16
0xf0 0x10 0xc2

# TSET bit, (addr8): [F0, addr, 0xA8+bit]
# CHECK: tset_dd8 4, 32
0xf0 0x20 0xac

# STCF bit, (addr8): [F0, addr, 0xA0+bit]
# CHECK: stcf_dd8 1, 64
0xf0 0x40 0xa1

# LDCF bit, (addr8): [F0, addr, 0x98+bit]
# CHECK: ldcf_dd8 6, 128
0xf0 0x80 0x9e

# ANDCF A, (addr8): [F0, addr, 0x28]
# CHECK: andcf_dd8 16
0xf0 0x10 0x28

# ORCF A, (addr8): [F0, addr, 0x29]
# CHECK: orcf_dd8 216
0xf0 0xd8 0x29

# XORCF A, (addr8): [F0, addr, 0x2A]
# CHECK: xorcf_dd8 7
0xf0 0x07 0x2a

# STCF A, (addr8): [F0, addr, 0x2C]
# CHECK: stcfa_dd8 32
0xf0 0x20 0x2c

# CALL (addr8): [F0, addr, 0x08]
# CHECK: call_dd8 8
0xf0 0x08 0x08

# JP cc, (addr8): [F0, addr, 0xD0+cc]
# CHECK: jp_dd8 8, 10
0xf0 0x0a 0xd8

# CHECK: jp_dd8 0, 255
0xf0 0xff 0xd0

# POP byte to (addr8): [F0, addr, 0x04]
# CHECK: popb_dd8 32
0xf0 0x20 0x04

# POP word to (addr8): [F0, addr, 0x06]
# CHECK: popw_dd8 32
0xf0 0x20 0x06

# Source 8-bit direct: LD rd, (addr8) — prefix C0
# CHECK: ld_sd8b a, 16
0xc0 0x10 0x21
